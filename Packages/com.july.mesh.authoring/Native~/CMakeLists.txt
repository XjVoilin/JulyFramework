cmake_minimum_required(VERSION 3.16)
project(JulyMeshOptimizer LANGUAGES CXX)
add_library(JulyMeshOptimizer SHARED bridge.cpp meshoptimizer/simplifier.cpp)
target_include_directories(JulyMeshOptimizer PRIVATE meshoptimizer)
target_compile_features(JulyMeshOptimizer PRIVATE cxx_std_11)
target_compile_definitions(JulyMeshOptimizer PRIVATE NDEBUG)
set_target_properties(JulyMeshOptimizer PROPERTIES CXX_VISIBILITY_PRESET hidden)
if(MSVC)
  set_property(TARGET JulyMeshOptimizer PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
