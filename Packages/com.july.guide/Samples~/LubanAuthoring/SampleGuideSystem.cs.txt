using System;
using System.Collections.Generic;
using System.Linq;
using July.Arch;
using July.Guide;
using guide_sample;

namespace July.Guide.Samples
{
    // Project-owned adapter. Rename/copy into the project; do not add it to July.Guide.Runtime.
    // Supply the generated Tables through your existing configuration loading path.
    public sealed class SampleGuideSystem : GuideSystemBase
    {
        private readonly Tables tables;
        public SampleGuideSystem(Tables tables) => this.tables = tables;

        protected override IEnumerable<GuideDefinition> CreateGuides()
        {
            var steps = tables.TbGuideStep.DataList.ToLookup(row => row.GuideId);
            foreach (var row in tables.TbGuide.DataList)
                yield return new GuideDefinition(row.Id, steps[row.Id].OrderBy(step => step.Sequence)
                    .Select(step => new GuideStepDefinition(step.StepId, (int)step.Kind, step.ParamId)).ToArray(),
                    row.Priority, row.CanSkip);
        }

        // This sample is explicitly triggered after its screen and target are ready.
        // Replace with project eligibility; the seed does not define a condition language.
        protected override bool CanStart(GuideDefinition guide) => true;

        protected override ProcedureBase CreateStepProcedure(GuideStepContext context)
        {
            var view = tables.TbGuideView.Get(context.Step.ParamId);
            var data = new GuideViewData(view.TextKey, confirmTextKey: view.ConfirmTextKey,
                skipTextKey: view.SkipTextKey);
            switch ((GuideStepKind)context.Step.Type)
            {
                case GuideStepKind.Confirm:
                    return new GuideConfirmProcedure(context, view.WindowId, data, view.TargetId);
                case GuideStepKind.ClickTarget:
                    return new GuideClickTargetProcedure(context, view.WindowId, view.TargetId, data);
                default:
                    throw new InvalidOperationException($"Unknown guide step type {context.Step.Type}.");
            }
        }

        // Override ResolveText here to call the project's existing localization service.
        // The unmodified sample deliberately displays the authored keys.
    }
}
